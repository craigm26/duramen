=== s14 C-2, s15 C-10, s16 C-9: an oracle that answers after 25 seconds
duramen 0.1
spec s 1
oracle node slow.mjs
op f
  input x? json
req A "a"
  example f {"x": 1}
    expect result.x = 1
--- slow.mjs
await new Promise((r) => setTimeout(r, 25000));
const chunks = [];
for await (const c of process.stdin) chunks.push(c);
for (const line of Buffer.concat(chunks).toString('utf8').split('\n')) {
  if (line.trim() === '') continue;
  const req = JSON.parse(line);
  console.log(JSON.stringify({ id: req.id, result: req.input }));
}
