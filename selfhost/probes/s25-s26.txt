// A probe's oracle: for each of the eight folders above it, whether it holds REGEN.json or package.json.
import fs from 'node:fs';
const up = (k, f) => fs.existsSync('../'.repeat(k) + f);
const chunks = [];
for await (const c of process.stdin) chunks.push(c);
for (const line of Buffer.concat(chunks).toString('utf8').split('\n')) {
  if (!line.trim()) continue;
  let r;
  try { r = JSON.parse(line); } catch { continue; }
  console.log(JSON.stringify({ id: r.id ?? null, result: { regen: [1, 2, 3, 4, 5, 6, 7, 8].map((k) => up(k, 'REGEN.json')), packageJson: [1, 2, 3, 4, 5, 6, 7, 8].map((k) => up(k, 'package.json')) } }));
}

import fs from 'node:fs';
