@op cases

--- order.mjs
// A probe's oracle: it counts its runs in a file beside it and answers with the run's number.
import fs from 'node:fs';
let n = 0;
try { n = Number(fs.readFileSync('runs.txt', 'utf8')); } catch {}
n += 1;
fs.writeFileSync('runs.txt', String(n));
const chunks = [];
for await (const c of process.stdin) chunks.push(c);
for (const line of Buffer.concat(chunks).toString('utf8').split('\n')) {
  if (!line.trim()) continue;
  let r;
  try { r = JSON.parse(line); } catch { continue; }
  console.log(JSON.stringify({ id: r.id ?? null, result: { run: n } }));
}

// A probe's oracle: on its first run it deletes flag.txt; each run reports whether flag.txt is there.
import fs from 'node:fs';
let n = 0;
try { n = Number(fs.readFileSync('runs.txt', 'utf8')); } catch {}
n += 1;
fs.writeFileSync('runs.txt', String(n));
const had = fs.existsSync('flag.txt');
if (n === 1 && had) fs.rmSync('flag.txt');
const chunks = [];
for await (const c of process.stdin) chunks.push(c);
for (const line of Buffer.concat(chunks).toString('utf8').split('\n')) {
  if (!line.trim()) continue;
  let r;
  try { r = JSON.parse(line); } catch { continue; }
  console.log(JSON.stringify({ id: r.id ?? null, result: { run: n, flag: had } }));
}
--- flag.txt
flag
